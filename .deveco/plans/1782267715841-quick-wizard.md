# 优化导入页面 UI/UX

## 改动清单

### 1. 来源选择区域改为卡片式

将"选择 HTML 文件"按钮和 URL 输入区域改为卡片式布局：

```typescript
Column({ space: 12 }) {
  // 从文件导入卡片
  Row({ space: 12 }) {
    SymbolGlyph($r('sys.symbol.doc'))
      .fontSize(24)
      .fontColor([$r('app.color.app_primary')])
    Column({ space: 4 }) {
      Text('从文件导入')
        .fontSize(15).fontWeight(FontWeight.Medium)
      Text('选择本地 HTML 文件')
        .fontSize(12).fontColor($r('app.color.app_text_muted'))
    }
    .alignItems(HorizontalAlign.Start)
  }
  .width('100%')
  .padding(16)
  .borderRadius(12)
  .backgroundColor($r('app.color.app_card_bg'))
  .border({ width: 1, color: $r('app.color.app_border') })
  .onClick(() => { this.pickHtmlFile() })

  // "或"分隔线
  Row({ space: 12 }) {
    Divider().layoutWeight(1).color($r('app.color.app_divider'))
    Text('或')
      .fontSize(12).fontColor($r('app.color.app_text_muted'))
    Divider().layoutWeight(1).color($r('app.color.app_divider'))
  }
  .width('90%')

  // 从网址导入卡片
  Column({ space: 8 }) {
    Row({ space: 12 }) {
      SymbolGlyph($r('sys.symbol.globe'))
        .fontSize(24)
        .fontColor([$r('app.color.app_primary')])
      Column({ space: 4 }) {
        Text('从网址导入')
          .fontSize(15).fontWeight(FontWeight.Medium)
        Text('输入 HTML 网址下载')
          .fontSize(12).fontColor($r('app.color.app_text_muted'))
      }
      .alignItems(HorizontalAlign.Start)
    }
    .width('100%')

    Row({ space: 8 }) {
      TextInput({ placeholder: 'https://example.com/page.html', text: this.urlInput })
        ...
      Button() { SymbolGlyph($r('sys.symbol.arrow_down_circle')) }
        ...
    }
  }
  .width('100%')
  .padding(16)
  .borderRadius(12)
  .backgroundColor($r('app.color.app_card_bg'))
  .border({ width: 1, color: $r('app.color.app_border') })
}
.width('90%')
```

### 2. 状态提示改为卡片

```typescript
if (this.statusMessage.length > 0) {
  Row({ space: 8 }) {
    SymbolGlyph(this.importMode === 'none' ? $r('sys.symbol.exclamationmark_circle') : $r('sys.symbol.checkmark_circle'))
      .fontSize(18)
      .fontColor([this.importMode === 'none' ? $r('app.color.app_error') : $r('app.color.app_success')])
    Text(this.statusMessage)
      .fontSize(14)
      .fontColor(this.importMode === 'none' ? $r('app.color.app_error') : $r('app.color.app_text_primary'))
  }
  .width('90%')
  .padding(12)
  .borderRadius(8)
  .backgroundColor(this.importMode === 'none' ? '#FEF2F2' : '#F0FDF4')
}
```
